#include <iostream>
#include <string>
using namespace std;

struct Node {
    string song;
    Node* next;
};

class Playlist {
private:
    Node* head;

public:
    Playlist() {
        head = nullptr;
    }

    void addSong(string name) {
        Node* newNode = new Node;
        newNode->song = name;
        newNode->next = nullptr;

        if (head == nullptr) {
            head = newNode;
        }
        else {
            Node* temp = head;

            while (temp->next != nullptr) {
                temp = temp->next;
            }

            temp->next = newNode;
        }

        cout << "Song added successfully!" << endl;
    }

    void removeSong(string name) {
        if (head == nullptr) {
            cout << "Playlist is empty" << endl;
            return;
        }

        if (head->song == name) {
            Node* temp = head;
            head = head->next;
            delete temp;

            cout << "Song removed successfully!" << endl;
            return;
        }

        Node* temp = head;

        while (temp->next != nullptr &&
               temp->next->song != name) {
            temp = temp->next;
        }

        if (temp->next == nullptr) {
            cout << "Song not found" << endl;
        }
        else {
            Node* deletedNode = temp->next;
            temp->next = deletedNode->next;
            delete deletedNode;

            cout << "Song removed successfully!" << endl;
        }
    }

    void show() {
        if (head == nullptr) {
            cout << "Playlist is empty" << endl;
            return;
        }

        Node* temp = head;
        int number = 1;

        while (temp != nullptr) {
            cout << number << ". " << temp->song << endl;
            temp = temp->next;
            number++;
        }
    }

    ~Playlist() {
        Node* temp = head;

        while (temp != nullptr) {
            Node* nextNode = temp->next;
            delete temp;
            temp = nextNode;
        }
    }
};

int main() {
    Playlist p;
    int choice;
    string name;

    do {
        cout << "\n1. Add song";
        cout << "\n2. Remove song";
        cout << "\n3. Show playlist";
        cout << "\n0. Exit";
        cout << "\nChoice: ";

        cin >> choice;

        if (choice == 1) {

            p.addSong(name);
        }
        else if (choice == 2) {
            cout << "Song name: ";
            getline(cin >> ws, name);
            p.removeSong(name);
        }
        else if (choice == 3) {
            p.show();
        }
        else if (choice == 0) {
            cout << "Goodbye!" << endl;
        }
        else {
            cout << "Invalid choice" << endl;
        }

    } while (choice != 0);

    return 0;
}
